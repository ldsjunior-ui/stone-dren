Envio do cálculo por e-mail

Ativar o envio em cinco minutos

O envio usa o Google Apps Script, que é gratuito. A ativação é feita uma vez, logado na conta Google que vai mandar os e-mails. Ninguém precisa da senha dessa conta: quem está logado cola o código e autoriza.

O que acontece depois de ativar: quando alguém pede o cálculo na calculadora de teste, o Google refaz a conta com a mesma tabela da página e envia o resultado para o e-mail da pessoa, com uma cópia para a conta que ativou.

Permissões pedidas pelo Google: enviar e-mail em nome da conta e ver o endereço de e-mail da conta (para saber para onde vai a cópia). Nada de acesso ao Drive, às planilhas ou aos e-mails recebidos.

Qual conta usar: quem recebe o cálculo vê essa conta como remetente e, se responder, a resposta vai para ela. Se preferir não mostrar um e-mail pessoal, crie uma conta Gmail só para a calculadora (por exemplo calculadora.stonedren@gmail.com), ative nela e escreva o e-mail do comercial em COPIA_PARA, no topo do código.

  1. Abra o Apps Script na conta certa

    Entre em script.google.com logado na conta que vai enviar os e-mails. Se o Google abrir outra conta, troque pela foto no canto de cima. Clique em Novo projeto.

  2. Dê um nome e cole o código

    Clique em Projeto sem título e escreva Stone Dren e-mail. No arquivo Código.gs (ou Code.gs), apague tudo o que estiver lá, clique em Copiar o código aqui embaixo e cole no lugar. Salve com o ícone de disquete (ou Ctrl+S, no Mac Cmd+S).

  3. Autorize com um teste

    Na barra de cima do editor, escolha a função testarEnvio e clique em Executar. O Google pede autorização:

    • Revisar permissões e escolha a mesma conta.
    • Aparece "O Google não verificou este app". É o normal para um script da própria conta. Clique em Avançado e depois em Acessar Stone Dren e-mail (não seguro).
    • Se aparecer uma lista com caixas de seleção, marque todas. Clique em Permitir (ou Continuar).

    Em alguns segundos chegam dois e-mails de teste na própria conta: o que a pessoa recebe e a cópia do comercial.

  4. Publique como app da Web

    Clique em Implantar e depois em Nova implantação (em inglês: Deploy, New deployment). Na engrenagem de "Selecionar tipo", escolha App da Web. Preencha assim:

    • Executar como: Eu (a própria conta).
    • Quem pode acessar: Qualquer pessoa, a última opção da lista. Não escolha "Somente eu" nem "Qualquer pessoa com uma Conta do Google": com essas, a calculadora não consegue chamar o envio.

    Clique em Implantar e copie a URL do app da Web. Ela começa com https://script.google.com/macros/s/ e termina em /exec.

  5. Confira numa janela anônima

    Abra uma janela anônima do navegador (no Chrome: Ctrl+Shift+N, no Mac Cmd+Shift+N), cole a URL e abra. Tem que aparecer uma linha de texto que começa com {"ok":true,"servico":"stone-dren-email".

    Se aparecer a tela de login do Google, o acesso ficou errado: volte em Implantar, Gerenciar implantações, clique no lápis, mude Quem pode acessar para Qualquer pessoa e clique em Implantar. A URL continua a mesma.

  6. Mande a URL para o Leonardo

    Ele coloca a URL na calculadora de teste e publica. Daí em diante, cada cálculo pedido na página chega por e-mail.

  var EPS = 1e-9;
  function ceilPack(kg, packKg) { return Math.ceil(kg / packKg - EPS); }
  function r2(n) { return Math.round(n * 100) / 100; }
  // Tira o ruído de ponto flutuante antes de arredondar: 18 x 0,15 x 1,15 dá 3,1049999999999995
  // em binário e viraria 3,1 na tela; com 12 dígitos significativos volta a ser 3,105 (e mostra 3,11).
  function limpo(n) { return typeof n === "number" && isFinite(n) && n !== 0 ? Number(n.toPrecision(12)) : n; }

  /* ---------- números em pt-BR ---------- */

  // Aceita "12,5", "12.5", "1.234,5", "1 234,5" e "1.500" (milhar). Devolve NaN se não for número.
  function parseNumero(v) {
    if (typeof v === "number") return isFinite(v) ? v : NaN;
    if (v == null) return NaN;
    var s = String(v).trim().replace(/\s+/g, "").replace(/m²|m2|m$/i, "");
    if (!s) return NaN;
    if (!/^[+]?[\d.,]+$/.test(s)) return NaN;
    var temPonto = s.indexOf(".") >= 0, temVirgula = s.indexOf(",") >= 0;
    if (temPonto && temVirgula) {
      if (s.lastIndexOf(",") > s.lastIndexOf(".")) s = s.replace(/\./g, "").replace(",", ".");
      else s = s.replace(/,/g, "");
    } else if (temVirgula) {
      if ((s.match(/,/g) || []).length > 1) return NaN;
      s = s.replace(",", ".");
    } else if (temPonto) {
      // "1.500" e "12.000.000" são milhar no Brasil; "1.5" e "12.25" são decimal
      if (/^\d{1,3}(\.\d{3})+$/.test(s)) s = s.replace(/\./g, "");
      else if ((s.match(/\./g) || []).length > 1) return NaN;
    }
    var n = Number(s);
    return isFinite(n) ? n : NaN;
  }

  // Arredondamento comercial em decimal (5,175 -> 5,18). toFixed sozinho erra
  // porque 5,175 em binário é 5,17499...; o deslocamento pelo expoente em texto não.
  function arred(n, casas) {
    var s = String(n);
    if (/e/i.test(s)) return Number(n.toFixed(casas));
    return Number(Math.round(Number(s + "e" + casas)) + "e-" + casas);
  }
  function fmt(n, casas) {
    if (casas == null) casas = 0;
    var neg = n < 0; n = Math.abs(n);
    var fixo = arred(n, casas).toFixed(casas);
    var partes = fixo.split(".");
    var inteiro = partes[0].replace(/\B(?=(\d{3})+(?!\d))/g, ".");
    return (neg ? "-" : "") + inteiro + (partes[1] ? "," + partes[1] : "");
  }
  // até 2 casas, sem zeros sobrando: 12 -> "12", 12,5 -> "12,5", 0,125 -> "0,13"
  function fmtAuto(n, max) {
    if (max == null) max = 2;
    var s = fmt(n, max);
    if (s.indexOf(",") >= 0) s = s.replace(/0+$/, "").replace(/,$/, "");
    return s;
  }
  // Nunca mostra "0,0 kg" para uma quantidade que existe: abaixo de 0,1 kg vira texto.
  function fmtKg(kg) {
    if (kg > 0 && kg < 0.1) return "menos de 0,1 kg";
    if (kg < 1) return fmtAuto(kg, 2) + " kg";
    return fmt(kg, kg < 100 ? 1 : 0) + " kg";
  }
  function fmtM3(m3) {
    if (m3 > 0 && m3 < 0.01) return "menos de 0,01 m³";
    return fmtAuto(m3, 2) + " m³";
  }
  function fmtCmDeMm(mm) { return fmtAuto(mm / 10, 1) + " cm"; }
  // Espessura da camada Stone Dren: sempre em mm, a mesma unidade do controle da página.
  function fmtMm(mm) { return fmt(mm) + " mm"; }
  // Contagem de embalagens com milhar e plural: 1 saco, 1.455.200 sacos.
  function qtd(n, singular, plural) { return fmt(n) + " " + (n === 1 ? singular : plural); }

  /* ---------- perfis ---------- */

  function perfil(uso) { return TABELA.perfis[uso] || null; }

  function graoRotulo(p) {
    return p.graoMinMm ? "de " + p.graoMinMm + " a " + p.graoMaxMm + " mm" : "até " + p.graoMaxMm + " mm";
  }
  // "pedra com grãos de até 6 mm" / "pedra com grãos de 2 a 4 mm" (sem o termo técnico granulometria)
  function graoFrase(p) {
    return "pedra com grãos " + (p.graoMinMm ? "de " + p.graoMinMm + " a " + p.graoMaxMm + " mm" : "de até " + p.graoMaxMm + " mm");
  }

  // Limites do controle de espessura de cada uso.
  function espessuraLimites(uso) {
    var p = perfil(uso);
    if (!p) return null;
    var pisoTecnico = TABELA.pedra.fatorGrao * p.graoMaxMm;
    var minimo = Math.max(p.espessuraMm, pisoTecnico);
    return {
      padraoMm: p.espessuraMm,
      minimoMm: minimo,                  // abaixo disso a página avisa
      sliderMinMm: Math.min(pisoTecnico, minimo),
      sliderMaxMm: Math.max(40, p.espessuraMm * 2),
      passoMm: 1
    };
  }

  function basesDoUso(uso) {
    var p = perfil(uso);
    if (!p) return [];
    var lista = [];
    ["solo", "contrapiso"].forEach(function (b) {
      lista.push({
        id: b,
        rotulo: (p.bases[b] && p.bases[b].rotulo) || TABELA.bases[b].rotulo,
        permitido: !!p.bases[b],
        motivo: p.baseBloqueada && p.baseBloqueada[b] || ""
      });
    });
    if (p.consulte && p.consulte.brita) lista.push({ id: "brita", rotulo: TABELA.bases.brita.rotulo, permitido: true, consulte: true, motivo: p.consulte.brita });
    return lista;
  }

  /* ---------- validação ---------- */

  // Devolve { ok, areaM2, erros:{campo: mensagem} } sem lançar exceção.
  function validarAmbiente(amb) {
    var erros = {};
    amb = amb || {};
    var L = TABELA.limites;
    var area = NaN;
    if (!perfil(amb.uso)) erros.uso = "Escolha o tipo de uso.";
    if (amb.modo === "medidas") {
      var c = parseNumero(amb.comprimento), l = parseNumero(amb.largura);
      var vazioC = amb.comprimento == null || String(amb.comprimento).trim() === "";
      var vazioL = amb.largura == null || String(amb.largura).trim() === "";
      if (vazioC) erros.comprimento = "Informe o comprimento em metros.";
      else if (isNaN(c)) erros.comprimento = "Use só números, por exemplo 4,5.";
      else if (!(c > 0)) erros.comprimento = "O comprimento precisa ser maior que zero.";
      else if (c > L.medidaMaxM) erros.comprimento = "Medida acima de " + fmt(L.medidaMaxM) + " m. Confira o número.";
      if (vazioL) erros.largura = "Informe a largura em metros.";
      else if (isNaN(l)) erros.largura = "Use só números, por exemplo 2,5.";
      else if (!(l > 0)) erros.largura = "A largura precisa ser maior que zero.";
      else if (l > L.medidaMaxM) erros.largura = "Medida acima de " + fmt(L.medidaMaxM) + " m. Confira o número.";
      if (!erros.comprimento && !erros.largura) {
        area = c * l;
        if (area > L.areaMaxM2) erros.largura = "A área passa de " + fmt(L.areaMaxM2) + " m². Para obras desse porte, fale com a equipe técnica.";
        else if (area < L.areaMinM2) erros.largura = "A área ficou menor que 0,01 m². Confira as medidas em metros.";
      }
    } else {
      var a = parseNumero(amb.area);
      if (amb.area == null || String(amb.area).trim() === "") erros.area = "Informe a área em m².";
      else if (isNaN(a)) erros.area = "Use só números, por exemplo 12,5.";
      else if (!(a > 0)) erros.area = "A área precisa ser maior que zero.";
      else if (a > L.areaMaxM2) erros.area = "A área passa de " + fmt(L.areaMaxM2) + " m². Para obras desse porte, fale com a equipe técnica.";
      else if (a < L.areaMinM2) erros.area = "Área menor que 0,01 m². Confira o número.";
      else area = a;
    }
    var ok = Object.keys(erros).length === 0;
    return { ok: ok, areaM2: ok ? area : null, erros: erros };
  }

  /* ---------- cálculo de um ambiente ---------- */

  // opts: { resina: "epoxi" | "pu" }
  function calcularAmbiente(amb, opts) {
    amb = amb || {};
    opts = opts || {};
    var resinaTipo = TABELA.resina.tipos[opts.resina] ? opts.resina : "epoxi";
    var p = perfil(amb.uso);
    var v = validarAmbiente(amb);
    var base = { id: amb.id != null ? amb.id : null, nome: String(amb.nome || "").trim(), uso: amb.uso };
    if (!v.ok) {
      base.estado = "incompleto";
      base.erros = v.erros;
      return base;
    }

    var area = v.areaM2;
    var lim = espessuraLimites(amb.uso);
    var baseId = amb.base === "solo" || amb.base === "contrapiso" || amb.base === "brita" ? amb.base : "contrapiso";
    var baseForcada = false, motivoBase = "";
    if (!p.bases[baseId] && !(p.consulte && p.consulte[baseId])) {
      motivoBase = (p.baseBloqueada && p.baseBloqueada[baseId]) || "Base não disponível para este uso.";
      baseId = "contrapiso";
      baseForcada = true;
    }

    var esp = parseNumero(amb.espessuraMm);
    var espAjustada = false;
    if (!(esp > 0)) esp = lim.padraoMm;
    if (esp < lim.sliderMinMm) { esp = lim.sliderMinMm; espAjustada = true; }
    if (esp > TABELA.limites.espessuraMaxMm) { esp = TABELA.limites.espessuraMaxMm; espAjustada = true; }
    esp = Math.round(esp);

    var r = {
      id: base.id, nome: base.nome, uso: amb.uso,
      usoRotulo: p.rotulo, usoDescricao: p.descricao,
      estado: "ok",
      aConfirmar: !!p.aConfirmar,
      areaM2: area,
      modo: amb.modo === "medidas" ? "medidas" : "area",
      base: baseId,
      baseRotulo: (p.bases[baseId] && p.bases[baseId].rotulo) || TABELA.bases[baseId].rotulo,
      baseForcada: baseForcada, motivoBase: motivoBase,
      espessuraMm: esp,
      espessuraPadraoMm: lim.padraoMm,
      espessuraMinimaMm: lim.minimoMm,
      abaixoDoMinimo: esp < lim.minimoMm,
      espessuraAjustada: espAjustada,
      graoMaxMm: p.graoMaxMm, graoMinMm: p.graoMinMm, graoRotulo: graoRotulo(p),
      resinaTipo: resinaTipo,
      pedra: null, resina: null, brita: null, laje: null, malha: null,
      caimentoPct: p.bases[baseId] && p.bases[baseId].caimentoPct || null,
      camadas: []
    };

    // Garagem direto sobre brita: estado nomeado, sem quantidades.
    if (p.consulte && p.consulte[baseId]) {
      r.estado = "consulte";
      r.motivo = p.consulte[baseId];
      r.camadas = [
        { id: "stone", rotulo: "Stone Dren", espessuraMm: esp, emEscala: true },
        { id: "brita", rotulo: "Brita", espessuraMm: null, emEscala: false },
        { id: "solo", rotulo: "Solo compactado", espessuraMm: null, emEscala: false }
      ];
      return r;
    }

    var T = TABELA;
    var pedraSemPerda = limpo(area * esp * T.pedra.kgPorM2PorMm);
    var pedraKg = limpo(pedraSemPerda * (1 + T.pedra.perda));
    r.pedra = {
      kg: pedraKg, kgSemPerda: pedraSemPerda, perdaPct: T.pedra.perda * 100,
      sacos: ceilPack(pedraKg, T.pedra.sacoKg), sacoKg: T.pedra.sacoKg, sacoProvisorio: T.pedra.sacoProvisorio
    };

    var pct = p.resinaPct[resinaTipo];
    var resinaKg = limpo(pedraKg * pct);
    r.resina = {
      tipo: resinaTipo, rotulo: T.resina.tipos[resinaTipo].rotulo, curto: T.resina.tipos[resinaTipo].curto, pct: pct,
      kg: resinaKg, kits: ceilPack(resinaKg, T.resina.kitKg), kitKg: T.resina.kitKg, kitProvisorio: T.resina.kitProvisorio
    };

    var cfgBase = p.bases[baseId] || {};
    if (baseId === "solo" && cfgBase.britaCm) {
      // em inteiros até a última divisão (area x cm x 115 / 10000) e sem o ruído binário
      var m3 = limpo(area * cfgBase.britaCm * Math.round(T.brita.empolamento * 100) / 10000);
      var kgB = limpo(m3 * T.brita.densidadeKgM3);
      r.brita = {
        cm: cfgBase.britaCm, m3: m3, kg: kgB, sacos: ceilPack(kgB, T.brita.sacoKg), sacoKg: T.brita.sacoKg,
        granel: m3 > T.brita.granelAcimaM3
      };
    }
    if (baseId === "solo" && cfgBase.laje) {
      r.laje = { cm: T.laje.cm, calculado: false, rotulo: T.laje.rotulo };
    }

/* Stone Dren · servidor do envio do cálculo por e-mail.
 *
 * Roda de graça no Google Apps Script, publicado como "app da Web" na conta que envia os
 * e-mails (Executar como: eu · Quem pode acessar: qualquer pessoa).
 *
 * O que faz a cada pedido da calculadora:
 *   1. confere o pedido (tamanho, origem, campo isca, id, nome, WhatsApp, cidade, e-mail);
 *   2. refaz a conta AQUI, com o mesmo motor da página (StoneDrenCalc), a partir do que a
 *      pessoa digitou. Número pronto vindo da página não é usado;
 *   3. envia o e-mail do cálculo para a pessoa e uma cópia para o comercial;
 *   4. guarda o id por 6 horas: o mesmo pedido repetido não gera outro e-mail.
 * Limites: pedidos por dia, pedidos por hora (todos somados), pedidos por e-mail por hora e a
 * cota diária do Gmail. O e-mail que vai para o endereço digitado não leva texto digitado.
 *
 * Funções que aparecem no menu "Executar" do editor:
 *   testarEnvio  manda um cálculo de exemplo para a própria conta (use uma vez para autorizar)
 *   doPost       recebe a calculadora (não rode à mão)
 *   doGet        responde se o serviço está no ar (abra a URL /exec no navegador); com ?id=
 *                diz o que aconteceu com aquele pedido (a página usa quando não conseguiu ler a
 *                resposta do envio: assim nunca diz "enviado" para um pedido que o servidor recusou)
 */

var VERSAO_SERVIDOR = "2026-09-24";
var PERFIS_ = ["Cliente final", "Aplicador", "Arquiteto ou construtor", "Interessado na franquia"];
var EMAIL_RE_ = /^[A-Za-z0-9._%+'-]+@[A-Za-z0-9-]+(\.[A-Za-z0-9-]+)*\.[A-Za-z]{2,24}$/;

var ID_RE_ = /^[a-f0-9]{16,64}$/;

function doGet(e) {
  // sem números de uso: a URL é pública e a cota do dia diria quantos pedidos a Stone Dren recebeu
  var r = { ok: true, servico: "stone-dren-email", versao: VERSAO_SERVIDOR, ligado: !!CONFIG.LIGADO, teste: !!CONFIG.MODO_TESTE };
  // o id é aleatório (96 bits) e só volta o resultado do envio, nunca nome, e-mail ou números
  var id = e && e.parameter ? String(e.parameter.id || "") : "";
  if (ID_RE_.test(id)) {
    var cache = CacheService.getScriptCache();
    var visto = cache.get("id:" + id), falha = cache.get("falha:" + id);
    if (visto) {
      var v = JSON.parse(visto);
      r.envio = { ok: true, cliente: !!v.cliente, copia: !!v.copia };
    } else if (falha) {
      r.envio = { ok: false, erro: falha };
    } else {
      r.envio = null;
    }
  }
  return responder_(r);
}

function doPost(e) {
  var ctx = { id: "" }, r;
  try {
    r = tratarPedido_(e && e.postData ? String(e.postData.contents || "") : "", ctx);
  } catch (err) {
    registrar_("falha interna", err);
    r = { ok: false, erro: "interno" };
  }
  // guarda também a recusa: a conferência por GET (?id=) conta à página o motivo de verdade
  if (!r.ok && ctx.id) {
    try { CacheService.getScriptCache().put("falha:" + ctx.id, String(r.erro || "interno"), 3600); } catch (e2) { /* segue */ }
  }
  return responder_(r);
}

  MailApp.sendEmail({ to: para, subject: mk.assunto, htmlBody: mk.html, body: mk.texto, name: CONFIG.NOME_REMETENTE });
  console.log("Pronto: dois e-mails de teste enviados para a própria conta. Envios restantes hoje: " + MailApp.getRemainingDailyQuota());
}

function tratarPedido_(texto, ctx) {
  ctx = ctx || {};
  if (!CONFIG.LIGADO) return { ok: false, erro: "desligado" };
  if (!texto || texto.length > 20000) return { ok: false, erro: "invalido" };
  var p;
  try { p = JSON.parse(texto); } catch (e) { return { ok: false, erro: "invalido" }; }
  if (!p || typeof p !== "object" || p.origem !== "calculadora-stone-dren") return { ok: false, erro: "invalido" };

  var id = String(p.id || "");
  if (!ID_RE_.test(id)) return { ok: false, erro: "invalido" };

  // campo isca preenchido = robô: responde como se tivesse enviado e não envia nada
  if (String(p.hp || p.site || "").trim()) return { ok: true, copia: true, cliente: true, id: id };
  ctx.id = id;

  var cache = CacheService.getScriptCache();
  var visto = cache.get("id:" + id);
  if (visto) return JSON.parse(visto);

  var L = p.lead || {};
  // o tamanho é conferido no que a pessoa digitou; se a limpeza esvaziar (ex.: nome só com emoji),
  // o pedido segue com "(não informado)" em vez de travar sem saída
  var nomeBruto = String(L.nome == null ? "" : L.nome).trim(), cidadeBruta = String(L.cidadeUf == null ? "" : L.cidadeUf).trim();
  var lead = {
    nome: nomeBruto.length >= 2 ? (limparTexto_(nomeBruto, 80) || "(não informado)") : "",
    whatsapp: String(L.whatsapp || "").replace(/\D/g, "").slice(0, 13),
    cidadeUf: cidadeBruta.length >= 2 ? (limparTexto_(cidadeBruta, 80) || "(não informado)") : "",
    perfil: PERFIS_.indexOf(String(L.perfil || "")) >= 0 ? String(L.perfil) : "",
    email: String(L.email || "").trim().toLowerCase(),
    optin: !!p.optinNovidades
  };
  if (lead.nome.length < 2 || lead.whatsapp.length < 10 || lead.cidadeUf.length < 2 || !lead.perfil) return { ok: false, erro: "invalido" };
  if (lead.email && !emailValido_(lead.email)) return { ok: false, erro: "email" };

  // a conta é refeita aqui, a partir do que foi digitado (nomes de ambiente sem link)
  var entrada = StoneDrenCalc.entradaDaObra(p.entrada || {});
  entrada.ambientes.forEach(function (a) { a.nome = limparTexto_(a.nome, 60); });
  var r = StoneDrenCalc.calcularObra(entrada);
  if (!r.totais && !r.consulte) return { ok: false, erro: "invalido" };

  // uma chave só: o MODO_TESTE do CONFIG (a página informa se é de teste, mas quem decide é o servidor)
  var teste = !!CONFIG.MODO_TESTE;
  // a página de origem é só um nome de arquivo (calculadora-teste.html): nada de texto livre aqui
  var pagina = /^[a-z0-9-]{1,40}\.html$/.test(String(p.pagina || "")) ? String(p.pagina) : "";
  var opts = opcoesEmail_(teste, pagina);
  var copia = copiaPara_();

  var trava = LockService.getScriptLock();
  if (!trava.tryLock(20000)) return { ok: false, erro: "ocupado" };
  try {
    visto = cache.get("id:" + id);
    if (visto) return JSON.parse(visto);

    var hoje = Utilities.formatDate(new Date(), "America/Sao_Paulo", "yyyy-MM-dd");
    var props = PropertiesService.getScriptProperties();
    var cont = lerContador_(props, hoje);
    if (cont.n >= CONFIG.LIMITE_POR_DIA) return { ok: false, erro: "cota" };
    var precisa = (lead.email ? 1 : 0) + (copia ? 1 : 0);
    if (MailApp.getRemainingDailyQuota() < Math.max(1, precisa)) return { ok: false, erro: "cota" };

    // teto por hora somando todos os pedidos: sem ele, uma rajada anônima queima a cota do dia inteiro
    var chaveGeral = "hh:" + Utilities.formatDate(new Date(), "America/Sao_Paulo", "yyyyMMddHH");
    var nGeral = parseInt(cache.get(chaveGeral) || "0", 10) || 0;
    if (nGeral >= CONFIG.LIMITE_POR_HORA) return { ok: false, erro: "hora" };
    var chaveHora = "", nHora = 0;
    if (lead.email) {
      chaveHora = "h:" + resumo_(caixaDoEmail_(lead.email));
      nHora = parseInt(cache.get(chaveHora) || "0", 10) || 0;
      if (nHora >= CONFIG.LIMITE_POR_EMAIL_HORA) return { ok: false, erro: "limite" };
    }
    cache.put(chaveGeral, String(nGeral + 1), 3600);
    if (chaveHora) cache.put(chaveHora, String(nHora + 1), 3600);

    var resp = { ok: true, cliente: false, copia: false, id: id };
    if (lead.email) {
      var mc = StoneDrenEmail.montarEmailCliente(r, lead, opts);
      var envioCliente = { to: lead.email, subject: umaLinha_(mc.assunto), htmlBody: mc.html, body: mc.texto, name: CONFIG.NOME_REMETENTE };
      if (copia) envioCliente.replyTo = copia;
      try {
        MailApp.sendEmail(envioCliente);
        resp.cliente = true;
      } catch (err) {
        registrar_("envio para a pessoa", err);
        return { ok: false, erro: /invalid|inválid/i.test(String(err && err.message)) ? "email" : "envio" };
      }
    }
    if (copia) {
      var mk = StoneDrenEmail.montarEmailComercial(r, lead, opts);
      var envioCopia = { to: copia, subject: umaLinha_(mk.assunto), htmlBody: mk.html, body: mk.texto, name: CONFIG.NOME_REMETENTE };
      if (lead.email) envioCopia.replyTo = lead.email;
      try {
        MailApp.sendEmail(envioCopia);
        resp.copia = true;
      } catch (err2) {
        registrar_("cópia para o comercial", err2);
        if (!resp.cliente) return { ok: false, erro: "envio" };
      }
    }
    if (!resp.cliente && !resp.copia) return { ok: false, erro: "envio" };
    // os e-mails já saíram: daqui em diante nada pode virar "não enviado" (nem gerar reenvio duplicado)
    try { cache.put("id:" + id, JSON.stringify(resp), 21600); } catch (e3) { registrar_("guardar o id", e3); }
    try { props.setProperty("contador", JSON.stringify({ dia: hoje, n: cont.n + 1 })); } catch (e4) { registrar_("contador do dia", e4); }
    return resp;
  } finally {
    trava.releaseLock();
  }
}

function opcoesEmail_(teste, pagina) {
  return {
    teste: !!teste,
    whatsapp: CONFIG.WHATSAPP_COMERCIAL,
    siteUrl: CONFIG.SITE_URL,
    empresa: CONFIG.EMPRESA || {},
    quando: Utilities.formatDate(new Date(), "America/Sao_Paulo", "dd/MM/yyyy 'às' HH:mm"),
    pagina: pagina || ""
  };
}

function copiaPara_() {
  var c = String(CONFIG.COPIA_PARA || "").trim();
  if (c && emailValido_(c)) return c;
  var eu = "";
  try { eu = Session.getEffectiveUser().getEmail(); } catch (e) { eu = ""; }
  return emailValido_(String(eu || "")) ? String(eu) : "";
}

function emailValido_(v) {
  return v.length <= 254 && EMAIL_RE_.test(v) && !/\.\.|^\.|\.@|@-/.test(v);
}

// Texto livre da pessoa (vai só para a cópia do comercial): lista do que é PERMITIDO, não do que é
// proibido. Fica letra, número, espaço e , . ( ) / ' -; @, :, pontos de outros alfabetos e espaços
// invisíveis caem no 1º passo. Depois, todo ponto colado entre letras ou números ganha um espaço
// ("golpe.com" vira "golpe. com", que o Gmail não transforma em link) em vez de apagar a palavra:
// assim "S.Paulo" ou "Ana.Paula" continuam legíveis. Só um número com um ponto (1.500) fica igual.
function limparTexto_(v, max) {
  var s = String(v == null ? "" : v).replace(/[^A-Za-z\u00C0-\u00D6\u00D8-\u00F6\u00F8-\u00FF0-9 ,.()\/'-]/g, " ");
  s = s.split(/\s+/).map(function (t) {
    if (/^\d+(\.\d+)?(,\d+)?$/.test(t)) return t;
    return t.replace(/^\.+/, "").replace(/([A-Za-z\u00C0-\u00D6\u00D8-\u00F6\u00F8-\u00FF0-9])\.(?=[A-Za-z\u00C0-\u00D6\u00D8-\u00F6\u00F8-\u00FF0-9])/g, "$1. ");
  }).join(" ");
  // sobras de endereço sem serventia num nome: http, https, www, ftp soltos e barras duplas
  s = s.replace(/\/{2,}/g, "/").replace(/(^|\s)(https?|www|ftp)(?=\s|$|\.)\.?/gi, " ");
  return s.replace(/\s+/g, " ").trim().slice(0, max || 80).trim();
}

// a mesma caixa de entrada escrita de jeitos diferentes conta como uma só no limite por hora:
// sem o +etiqueta em qualquer domínio e, no Gmail, sem os pontos antes do @
function caixaDoEmail_(email) {
  var partes = String(email || "").toLowerCase().split("@");
  if (partes.length !== 2) return String(email || "").toLowerCase();
  var local = partes[0].split("+")[0], dom = partes[1];
  if (dom === "gmail.com" || dom === "googlemail.com") { local = local.replace(/\./g, ""); dom = "gmail.com"; }
  return local + "@" + dom;
}

function umaLinha_(s) { return String(s || "").replace(/[\r\n]+/g, " ").slice(0, 200); }

function resumo_(s) {
  var b = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(s), Utilities.Charset.UTF_8);
  return b.map(function (x) { return ("0" + ((x + 256) % 256).toString(16)).slice(-2); }).join("").slice(0, 32);
}

function lerContador_(props, hoje) {
  var c = null;
  try { c = JSON.parse(props.getProperty("contador") || "null"); } catch (e) { c = null; }
  return c && c.dia === hoje ? { dia: hoje, n: parseInt(c.n, 10) || 0 } : { dia: hoje, n: 0 };
}

function responder_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

// registro para o painel "Execuções" do Apps Script: só a etapa e a mensagem do erro, nenhum dado da pessoa
function registrar_(etapa, err) {
  var msg = String(err && err.message || err).replace(/\S+@\S+/g, "[e-mail]").slice(0, 300);
  try { console.error("Stone Dren · " + etapa + ": " + msg); } catch (e) { /* segue */ }
}
